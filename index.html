Drop-in SDK

import { research } from 'https://sjgant80-hub.github.io/fall-raas/fall-raas-sdk.js';

const result = await research({
  question: "What's the actual evidence for the 4-day work week?",
  angles: 5,           // 1-7 · default 5
  votes: 3,            // 2-5 · adversarial panel size
  killThreshold: 2,    // votes needed to kill a claim
  byoKey: 'sk-ant-...',  // stored in IndexedDB · never sent anywhere else
  onProgress: (e) => console.log(e),
});

// → { summary, findings, refuted, openQuestions, citations, stats }
console.log(result.refuted);  // the hallucination kill-list
